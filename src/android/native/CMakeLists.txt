# src/android/native/CMakeLists.txt
#
# Native libraries for the Android build of formuls, cross-compiled once per
# ABI by build-android.sh with the NDK's CMake toolchain file:
#
#   libformulspd.a        libpd (with the Pd "extra" objects and the queued
#                         messaging utils) PLUS every external the patch
#                         needs, linked in statically. The JUCE app links
#                         this into libjuce_jni.so.
#
#   libformuls_node.so    a small JNI shim that runs Open Stage Control on
#                         the nodejs-mobile runtime (libnode.so). It is loaded
#                         by GuiServerService in the app's separate ":gui"
#                         process, never by the audio process.
#
# Why externals are static on Android
# -----------------------------------
# On macOS and Linux the externals are .pd_darwin / .pd_linux shared objects
# that Pd dlopen()s from pd/externals at patch load time. An Android app can
# only load native code from its own lib/ directory, and libpd's CMake build
# does not even enable dlopen support there. So the three Faust objects and
# abl_link~ are compiled into libformulspd.a, and
# formuls_setup_static_externals() registers them with Pd before _main.pd is
# opened (see FormulsEngine.cpp). To the patch they look exactly as if they
# had been loaded from pd/externals.
#
# Inputs (passed with -D by build-android.sh):
#   FORMULS_FAUST_GEN_DIR   folder holding formuls.cpp, f_reverb.cpp and
#                           f_repeater.cpp, generated by
#                           "faust -i -a puredata.cpp ..."
#   NODEJS_MOBILE_DIR       an unpacked nodejs-mobile Android release
#                           (include/node, bin/<abi>/libnode.so)

cmake_minimum_required (VERSION 3.25)
project (formuls_android_native C CXX)

set (CMAKE_CXX_STANDARD 17)
set (CMAKE_CXX_STANDARD_REQUIRED ON)
set (CMAKE_POSITION_INDEPENDENT_CODE ON)

set (FORMULS_SRC   "${CMAKE_CURRENT_LIST_DIR}/../..")
set (LIBPD_DIR     "${FORMULS_SRC}/libs/libpd")
set (ABL_LINK_DIR  "${FORMULS_SRC}/libs/abl_link/external")

if (NOT FORMULS_FAUST_GEN_DIR)
    message (FATAL_ERROR "Set FORMULS_FAUST_GEN_DIR to the folder of faust-generated .cpp files")
endif()

if (NOT NODEJS_MOBILE_DIR)
    message (FATAL_ERROR "Set NODEJS_MOBILE_DIR to an unpacked nodejs-mobile Android release")
endif()

#==============================================================================
# libpd, built by its own CMake project (static only)
set (PD_UTILS      ON  CACHE BOOL "" FORCE)
set (PD_EXTRA      ON  CACHE BOOL "" FORCE)
set (PD_MULTI      OFF CACHE BOOL "" FORCE)
set (LIBPD_STATIC  ON  CACHE BOOL "" FORCE)
set (LIBPD_SHARED  OFF CACHE BOOL "" FORCE)
add_subdirectory ("${LIBPD_DIR}" libpd EXCLUDE_FROM_ALL)

#==============================================================================
# The externals the patch uses, compiled straight into the library.
set (FAUST_EXTERNALS formuls f_reverb f_repeater)
set (FAUST_SOURCES)

foreach (dsp IN LISTS FAUST_EXTERNALS)
    set (src "${FORMULS_FAUST_GEN_DIR}/${dsp}.cpp")

    if (NOT EXISTS "${src}")
        message (FATAL_ERROR "Missing faust-generated source: ${src}")
    endif()

    list (APPEND FAUST_SOURCES "${src}")

    # Each generated file declares a class called "mydsp". faust2puredata
    # renames it per object with -Dmydsp=<name>; doing the same here keeps
    # the three objects apart when they end up in one library, and gives
    # their setup functions the names Pd expects (formuls_tilde_setup...).
    #
    # puredata.cpp's PdUI class also defines its methods outside the class
    # body, so as separate .pd_* files each had its own copy but in one
    # library they collide at link time. It gets the same per-object rename.
    #
    # The prefix header is force-included ahead of the generated code; see
    # faust_android_prefix.h for why.
    set_source_files_properties ("${src}" PROPERTIES
        COMPILE_OPTIONS "-Dmydsp=${dsp};-DPdUI=PdUI_${dsp};-DNEWIO=0;-include;${CMAKE_CURRENT_LIST_DIR}/faust_android_prefix.h;-Wno-unused-parameter;-Wno-unused-function;-ffast-math")
endforeach()

add_library (formulspd STATIC
    formuls_static_externals.c
    ${FAUST_SOURCES}
    "${ABL_LINK_DIR}/abl_link~.cpp"
    "${ABL_LINK_DIR}/abl_link_instance.cpp")

target_include_directories (formulspd PRIVATE
    "${LIBPD_DIR}/pure-data/src"
    "${ABL_LINK_DIR}/link/include"
    "${ABL_LINK_DIR}/link/modules/asio-standalone/asio/include")

# Same definitions as abl_link's own Android recipe
# (src/libs/abl_link/android/CMakeLists.txt). Android is Linux as far as
# Ableton Link is concerned; bionic has had getifaddrs() since API 24, below
# the app's minimum of 28, so the bundled android-ifaddrs shim is not needed.
target_compile_definitions (formulspd PRIVATE
    LINK_PLATFORM_LINUX=1
    ABL_LINK_OFFSET_MS=23)

set_source_files_properties ("${ABL_LINK_DIR}/abl_link~.cpp"
                             "${ABL_LINK_DIR}/abl_link_instance.cpp"
    PROPERTIES COMPILE_OPTIONS "-fexceptions;-Wno-multichar")

# libpd_static carries PD=1 and friends as public definitions, so linking it
# here also gives the externals the same view of m_pd.h as libpd itself.
# The JUCE project links the two archives in the order "formulspd;pd"
# (libformulspd.a, then libpd.a), because the externals call into libpd.
target_link_libraries (formulspd PUBLIC libpd_static)

#==============================================================================
# JNI shim that starts nodejs-mobile in the :gui process.
add_library (node SHARED IMPORTED)
set_target_properties (node PROPERTIES
    IMPORTED_LOCATION "${NODEJS_MOBILE_DIR}/bin/${ANDROID_ABI}/libnode.so")

add_library (formuls_node SHARED formuls_node_jni.cpp)
target_include_directories (formuls_node PRIVATE "${NODEJS_MOBILE_DIR}/include/node")
target_link_libraries (formuls_node PRIVATE node log)
